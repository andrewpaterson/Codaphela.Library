	if (shifted == 0x10000000000LL && mask == 0x0FFFFFFFFFFFFFFFull && precedence == 15 && (1LL << (seven * 5 + 5)) == shifted && (1 + two * 3 << 1 & 0xFE | 1) == precedence) r += 512;
	return r;
}
